      0 < neutralNF661Quadratic H z) :
    Function.Injective (neutralNF606Apply H) := by
  classical
  intro x y hxy
  let z : Fin n → ℝ := fun j => x j - y j
  have hzrow (i : Fin n) : neutralNF606Apply H z i = 0 := by
    dsimp only [z]
    rw [neutralNF606Apply_sub, congrFun hxy i, sub_self]
  have hq : neutralNF661Quadratic H z = 0 := by
    rw [neutralNF668Quadratic_eq_apply]
    apply Finset.sum_eq_zero
    intro i _
    change neutralNF606Apply H z i * z i = 0

    by_contra hne
